Integrasi Google Spreadsheet

Koneksikan formulir ini langsung ke Google Sheets Anda

Langkah-Langkah Integrasi:

  1. Buka Google Sheets baru di browser Anda.
  2. Klik menu Ekstensi (Extensions) > pilih Apps Script.
  3. Hapus semua kode bawaan, lalu **salin kode di bawah ini** dan paste ke editor Apps Script.
  4. Klik **Simpan (Save/Ctrl+S)**, lalu klik tombol **Terapkan (Deploy)** > **Penerapan Baru (New Deployment)**.
  5. Pilih Jenis: **Aplikasi Web (Web App)**.
  6. Atur Who has access / Siapa yang memiliki akses menjadi Siapa saja (Anyone).
  7. Klik **Terapkan (Deploy)**, izinkan akses Google Account Anda, lalu **salin Web App URL** yang didapatkan.
  8. Tempelkan Web App URL tersebut pada kolom di bawah lalu klik **Simpan & Hubungkan**.
Kode Google Apps Script: 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Baris Header jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Waktu Pengiriman", 
        "Nama Lengkap / Badan Publik", 
        "No. KTP / SIM / Paspor", 
        "No. Pengesahan Legalitas", 
        "Alamat Lengkap", 
        "Pekerjaan", 
        "No. Telepon / WA", 
        "Email", 
        "Jenis Informasi", 
        "Informasi yang Dibutuhkan", 
        "Cara Memperoleh Informasi", 
        "Cara Mendapatkan Salinan", 
        "Tujuan Permohonan Informasi", 
        "Status File KTP", 
        "Status File Kemenkumham", 
        "Status File Kesbangpol", 
        "Status File Kartu Pers"
      ]);
      sheet.getRange(1, 1, 1, 17).setFontWeight("bold").setBackground("#e2e8f0");
    }
    
    // Tambahkan data baris baru
    sheet.appendRow([
      new Date(),
      data.nama_lengkap || "",
      data.no_ktp || "",
      data.no_pengesahan || "",
      data.alamat || "",
      data.pekerjaan || "",
      data.no_hp || "",
      data.email || "",
      data.jenis_informasi || "",
      data.informasi_dibutuhkan || "",
      data.cara_memperoleh || "",
      data.cara_salinan || "",
      data.tujuan_permohonan || "",
      data.file_ktp || "-",
      data.file_kemenkumham || "-",
      data.file_kesbangpol || "-",
      data.file_pers || "-"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
